# run_experiment.py
"""
Refactored script for training a simple MLP on MNIST to demonstrate
hyperparameter transfer with and without muP.

This script incorporates software engineering best practices:
- Argument parsing for configuration.
- Systematic logging to console and file.
- Model checkpointing for saving and resuming training.
- Organized saving of results (metrics and plots).

To run a single experiment:
$ python run_experiment.py --width 64 --lr 0.01 --optimizer adam --log_dir results/adam

To replicate the original hyperparameter sweep using a shell script:
(See the accompanying sweep.sh file)
"""
import argparse
import logging
import os
import sys
from typing import List, Tuple

import matplotlib.pyplot as plt
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from keras.datasets import mnist
from torch.utils.data import DataLoader, TensorDataset

# ===================================================================================
# 1. Model and Optimizer Definitions (from original notebook)
# ===================================================================================

class MLP(nn.Module):
    """A simple Multi-Layer Perceptron."""
    def __init__(self, input_size: int = 784, hidden_sizes: List[int] = [64, 64, 64], num_classes: int = 10):
        super(MLP, self).__init__()
        all_hidden_sizes = [input_size] + hidden_sizes + [num_classes]
        self.layers = nn.ModuleList()
        for i in range(len(all_hidden_sizes) - 1):
            self.layers.append(nn.Linear(all_hidden_sizes[i], all_hidden_sizes[i+1]))
        self.sigmoid = nn.Sigmoid()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = x.view(x.size(0), -1)
        for layer in self.layers[:-1]:
            x = self.sigmoid(layer(x))
        x = self.layers[-1](x)
        return x

class SimpleAdam(optim.Optimizer):
    """Custom Adam implementation from notebook."""
    def __init__(self, params, lr: float = 1e-1, b1: float = 0.9, b2: float = 0.999):
        defaults = dict(lr=lr, b1=b1, b2=b2)
        super(SimpleAdam, self).__init__(params, defaults)

    @torch.no_grad()
    def step(self, closure=None):
        for group in self.param_groups:
            for p in group['params']:
                if p.grad is None:
                    continue
                grad = p.grad.data
                state = self.state[p]
                if len(state) == 0:
                    state["step"] = torch.tensor(0.0)
                    state['momentum'] = torch.zeros_like(p)
                    state['variance'] = torch.zeros_like(p)

                state['step'] += 1
                m, v = state['momentum'], state['variance']
                m.lerp_(grad, 1 - group["b1"])
                v.lerp_(grad**2, 1 - group["b2"])

                m_hat = m / (1 - group["b1"]**state['step'])
                v_hat = v / (1 - group["b2"]**state['step'])
                u = m_hat / (torch.sqrt(v_hat) + 1e-16)
                p.add_(u, alpha=-group['lr'])

class SimpleAdamMuP(optim.Optimizer):
    """Custom muP Adam implementation from notebook."""
    def __init__(self, params, lr: float = 1e-1, b1: float = 0.9, b2: float = 0.999):
        defaults = dict(lr=lr, b1=b1, b2=b2)
        super(SimpleAdamMuP, self).__init__(params, defaults)

    @torch.no_grad()
    def step(self, closure=None):
        for group in self.param_groups:
            for p in group['params']:
                if p.grad is None:
                    continue
                grad = p.grad.data
                state = self.state[p]
                if len(state) == 0:
                    state["step"] = torch.tensor(0.0)
                    state['momentum'] = torch.zeros_like(p)
                    state['variance'] = torch.zeros_like(p)

                state['step'] += 1
                m, v = state['momentum'], state['variance']
                m.lerp_(grad, 1 - group["b1"])
                v.lerp_(grad**2, 1 - group["b2"])

                m_hat = m / (1 - group["b1"]**state['step'])
                v_hat = v / (1 - group["b2"]**state['step'])
                u = m_hat / (torch.sqrt(v_hat) + 1e-16)

                lr = group['lr']
                # --- muP scaling logic ---
                if p.dim() > 1: # Weight matrix
                    d_in = p.shape[1]
                    lr = lr / d_in
                # --- End muP scaling ---

                p.add_(u, alpha=-lr)


# ===================================================================================
# 2. Refactored Helper Functions
# ===================================================================================

def setup_logging(log_dir: str, run_name: str):
    """Configures the logging module to output to console and file."""
    # This setup allows for structured logging, which is more powerful than print().
    # For advanced use cases, one might integrate with tools like Weights & Biases (wandb).
    # Citation: Python `logging` HOWTO - https://docs.python.org/3/howto/logging.html
    log_path = os.path.join(log_dir, f"{run_name}.log")
    os.makedirs(log_dir, exist_ok=True)
    
    logger = logging.getLogger()
    logger.setLevel(logging.INFO)
    
    # Remove default handlers
    if logger.hasHandlers():
        logger.handlers.clear()

    # Console handler
    ch = logging.StreamHandler(sys.stdout)
    ch.setFormatter(logging.Formatter('[%(asctime)s] %(message)s', datefmt='%H:%M:%S'))
    logger.addHandler(ch)

    # File handler
    fh = logging.FileHandler(log_path)
    fh.setFormatter(logging.Formatter('%(asctime)s - %(levelname)s - %(message)s'))
    logger.addHandler(fh)
    
    logging.info(f"Logging initialized. Log file at: {log_path}")

def get_data_loaders(batch_size: int) -> Tuple[DataLoader, DataLoader]:
    """Loads MNIST data and creates PyTorch DataLoaders."""
    (train_images, train_labels), (valid_images, valid_labels) = mnist.load_data()

    # Normalize and convert to PyTorch tensors
    train_images = torch.from_numpy(train_images.astype(np.float32) / 255.0)
    train_labels = torch.from_numpy(train_labels).long()
    valid_images = torch.from_numpy(valid_images.astype(np.float32) / 255.0)
    valid_labels = torch.from_numpy(valid_labels).long()

    # Using TensorDataset and DataLoader is standard practice for batching and shuffling.
    # Citation: PyTorch `torch.utils.data` documentation.
    train_dataset = TensorDataset(train_images, train_labels)
    valid_dataset = TensorDataset(valid_images, valid_labels)
    
    train_loader = DataLoader(dataset=train_dataset, batch_size=batch_size, shuffle=True)
    valid_loader = DataLoader(dataset=valid_dataset, batch_size=batch_size, shuffle=False)
    
    return train_loader, valid_loader

def save_checkpoint(state: dict, is_best: bool, checkpoint_dir: str, filename: str = 'checkpoint.pth.tar'):
    """Saves model and training state."""
    # Checkpointing is critical for long training jobs to recover from interruptions.
    # Citation: PyTorch official examples - imagenet/main.py
    os.makedirs(checkpoint_dir, exist_ok=True)
    filepath = os.path.join(checkpoint_dir, filename)
    torch.save(state, filepath)
    if is_best:
        best_filepath = os.path.join(checkpoint_dir, 'model_best.pth.tar')
        torch.save(state, best_filepath)
        logging.info(f"Saved new best model to {best_filepath}")


# ===================================================================================
# 3. Main Training and Evaluation Logic
# ===================================================================================

def train(args: argparse.Namespace):
    """Main function to run a single training experiment."""
    
    # --- Setup ---
    run_name = f"w{args.width}_lr{args.lr}_opt{args.optimizer}_seed{args.seed}"
    setup_logging(args.log_dir, run_name)
    
    use_cuda = not args.no_cuda and torch.cuda.is_available()
    device = torch.device("cuda" if use_cuda else "cpu")
    torch.manual_seed(args.seed)
    np.random.seed(args.seed)
    logging.info(f"Using device: {device}")
    logging.info(f"Starting run: {run_name}")
    logging.info(f"Arguments: {vars(args)}")

    # --- Data ---
    train_loader, valid_loader = get_data_loaders(args.batch_size)

    # --- Model and Optimizer ---
    hidden_layers = [args.width] * 3 # 3 hidden layers of the same width
    model = MLP(hidden_sizes=hidden_layers).to(device)
    
    optimizer_map = {
        'adam': SimpleAdam,
        'mup': SimpleAdamMuP
    }
    optimizer_class = optimizer_map.get(args.optimizer.lower())
    if optimizer_class is None:
        raise ValueError("Invalid optimizer specified. Choose 'adam' or 'mup'.")
    optimizer = optimizer_class(model.parameters(), lr=args.lr)
    
    criterion = nn.CrossEntropyLoss()
    
    # --- Training Loop ---
    best_loss = float('inf')
    for epoch in range(1, args.epochs + 1):
        # Train for one epoch
        model.train()
        for batch_idx, (data, target) in enumerate(train_loader):
            data, target = data.to(device), target.to(device)
            optimizer.zero_grad()
            output = model(data)
            loss = criterion(output, target)
            loss.backward()
            optimizer.step()
        
        # Evaluate on validation set
        model.eval()
        valid_loss = 0
        with torch.no_grad():
            for data, target in valid_loader:
                data, target = data.to(device), target.to(device)
                output = model(data)
                valid_loss += criterion(output, target).item()
        
        valid_loss /= len(valid_loader)
        logging.info(f'Epoch: {epoch}/{args.epochs}\tValidation Loss: {valid_loss:.6f}')
        
        # --- Checkpointing ---
        is_best = valid_loss < best_loss
        best_loss = min(valid_loss, best_loss)
        save_checkpoint({
            'epoch': epoch,
            'state_dict': model.state_dict(),
            'optimizer': optimizer.state_dict(),
            'best_loss': best_loss,
            'args': args
        }, is_best, checkpoint_dir=args.checkpoint_dir, filename=f"{run_name}.pth.tar")

    # --- Save Final Results ---
    results_path = os.path.join(args.log_dir, 'results.csv')
    is_new_file = not os.path.exists(results_path)
    with open(results_path, 'a') as f:
        if is_new_file:
            f.write("width,lr,optimizer,seed,final_loss\n")
        f.write(f"{args.width},{args.lr},{args.optimizer},{args.seed},{best_loss}\n")
    logging.info(f"Final validation loss: {best_loss:.6f}. Results saved to {results_path}")


# ===================================================================================
# 4. Argparse and Main Execution Block
# ===================================================================================

if __name__ == '__main__':
    # Using argparse makes the script configurable from the command line,
    # facilitating automated sweeps and improving reproducibility.
    # Citation: Python `argparse` documentation - https://docs.python.org/3/library/argparse.html
    parser = argparse.ArgumentParser(description='MLP Hyperparameter Transfer Experiment')
    
    # Model parameters
    parser.add_argument('--width', type=int, default=64, help='hidden layer width')
    
    # Training parameters
    parser.add_argument('--optimizer', type=str, default='adam', choices=['adam', 'mup'], help='optimizer to use')
    parser.add_argument('--lr', type=float, default=0.01, help='learning rate')
    parser.add_argument('--batch-size', type=int, default=64, help='input batch size for training')
    parser.add_argument('--epochs', type=int, default=20, help='number of epochs to train (increased from notebook for stability)')
    
    # Environment and reproducibility
    parser.add_argument('--no-cuda', action='store_true', default=False, help='disables CUDA training')
    parser.add_argument('--seed', type=int, default=42, help='random seed for reproducibility')
    
    # Logging and saving
    parser.add_argument('--log-dir', type=str, default='./logs', help='directory to save logs and results')
    parser.add_argument('--checkpoint-dir', type=str, default='./checkpoints', help='directory to save model checkpoints')
    
    args = parser.parse_args()
    
    try:
        train(args)
    except Exception as e:
        logging.error(f"An error occurred during training: {e}", exc_info=True)
        sys.exit(1)

#!/bin/bash

# This script automates the hyperparameter sweep, demonstrating how a command-line
# interface enables robust, repeatable experimentation.

echo "Starting Adam Sweep..."
ADAM_LRS=(0.0001 0.0003 0.001 0.003 0.01 0.03 0.1)
WIDTHS=(16 32 64 128 256)

for width in "${WIDTHS[@]}"; do
  for lr in "${ADAM_LRS[@]}"; do
    echo "Running Adam with width=${width} and lr=${lr}"
    python run_experiment.py --optimizer adam --width "$width" --lr "$lr" --log-dir "results/adam" --checkpoint-dir "checkpoints/adam"
  done
done

echo "Starting muP Sweep..."
MUP_LRS=(0.03 0.1 0.3 1.0 3.0)

for width in "${WIDTHS[@]}"; do
  for lr in "${MUP_LRS[@]}"; do
    echo "Running muP with width=${width} and lr=${lr}"
    python run_experiment.py --optimizer mup --width "$width" --lr "$lr" --log-dir "results/mup" --checkpoint-dir "checkpoints/mup"
  done
done

echo "Sweep complete. Check the 'results/' directory."
