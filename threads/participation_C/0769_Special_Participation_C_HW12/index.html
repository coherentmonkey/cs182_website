"""
maml_refactored.py

Refactored teaching code for:
    - MAML-style binary classification

This module is designed to be used alongside the CS182 MAML notebook.
It improves structure, documentation, and style without changing the
underlying math or assignment logic.

You are expected to already have the following defined in your notebook
or another module:
    - generate_x(n, distribution)
    - featurize(x, phi_type, d, normalize)
    - generate_y(features, k_idx, k_val)
    - add_label_noise(y, noise_prob)
    - my_sign(y)
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Sequence, Tuple

import logging
import random

import numpy as np
import torch
from torch import Tensor, nn

# If you use "higher" for MAML inner-loop optimization, ensure it's installed:
#   pip install higher

    import higher
except ImportError:  # optional dependency for this module
    higher = None  # type: ignore

def set_global_seed(seed: int) -> None:
    """Set seeds for Python, NumPy and PyTorch for reproducible experiments.

    Args:
        seed: Integer seed value.
    """
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)

@dataclass
class MetaClassificationConfig:
    """Configuration for MAML-style binary classification.

    Attributes:
        n_meta_train: Number of meta-training examples per task.
        feature_type: Name of the feature mapping φ(x).
        d: Dimensionality of the feature space.
        k_idx: Indices of non-zero teacher weights.
        k_val: Values of the non-zero teacher weights.
        label_noise_prob: Probability of flipping each label.
    """

    n_meta_train: int
    feature_type: str
    d: int
    k_idx: Sequence[int]
    k_val: Sequence[float]
    label_noise_prob: float


def classification_logistic_loss(logits: Tensor, y: Tensor) -> Tensor:
    """Binary logistic loss for labels in ℝ (thresholded to {+1, -1}).

    The targets y are first mapped to z ∈ {+1, -1} using my_sign. The loss is:

        L = mean_i log(1 + exp(- z_i * f_i)),

    where f_i are the logits.

    Args:
        logits: Model outputs f(x) of shape (N, 1) or (N,).
        y: Real-valued targets (N, 1) or (N,) to be mapped to {+1, -1}.

    Returns:
        Scalar tensor mean loss over the batch.
    """
    z = my_sign(y).float()  # type: ignore[name-defined]
    loss = torch.mean(torch.log1p(torch.exp(-z * logits)))
    return loss


def meta_update_classification(
    w_t: Tensor,
    feature_weights_t: Tensor,
    config: MetaClassificationConfig,
) -> Tensor:
    """Compute meta-training loss for a binary classification task.

    This implements the negative log-likelihood for a logistic model:

        L(α) = sum_i log(1 + exp(- z_i f_i)),
        z_i ∈ {+1, -1}, f_i = ⟨α, φ(x_i)⟩.

    The function assumes that the following helper functions are defined in
    your environment (as in the CS182 MAML notebook):

        - generate_x(n, distribution)
        - featurize(x, phi_type, d, normalize)
        - generate_y(features, k_idx, k_val)
        - add_label_noise(y, noise_prob)
        - my_sign(y)

    Args:
        w_t: Current meta-parameters α_t, shape (d, 1).
        feature_weights_t: Included for API symmetry with regression;
            not used here but kept for teaching purposes.
        config: Meta-learning configuration (features, label noise, etc.).

    Returns:
        Scalar tensor containing the meta-loss for this task.
    """
    # Generate meta-training inputs and features
    x_meta = generate_x(config.n_meta_train, distribution="uniform_random")  # type: ignore[name-defined]
    features_meta = featurize(  # type: ignore[name-defined]
        x_meta, phi_type=config.feature_type, d=config.d, normalize=True
    )
    features_meta_t = torch.as_tensor(features_meta, dtype=torch.float32)

    # Generate clean labels from the teacher and add label noise
    y_meta = generate_y(features_meta, config.k_idx, config.k_val)  # type: ignore[name-defined]
    y_meta = add_label_noise(y_meta, config.label_noise_prob)  # type: ignore[name-defined]
    y_meta_t = torch.as_tensor(y_meta, dtype=torch.float32)

    # Model logits f(x) = φ(x)^T α
    y_meta_pred_t = features_meta_t @ w_t

    # Convert real-valued targets to {+1, -1}
    z_meta_t = my_sign(y_meta_t).float()  # type: ignore[name-defined]

    # Logistic loss: sum_i log(1 + exp(-z_i f_i))
    loss = torch.sum(torch.log1p(torch.exp(-z_meta_t * y_meta_pred_t)))

    return loss


def meta_learning_classification(
    dm: nn.Module,
    opt_inner: torch.optim.Optimizer,
    tasks: Sequence[Tuple[Tensor, Tensor]],
    num_gd_steps: int,
) -> Tensor:
    """Perform inner-loop adaptation over a sequence of classification tasks.

    This function is a *template* for the inner loop shown in the MAML
    classification notebook. It uses the same logistic loss as the
    meta-update function and is written to preserve teaching value.

    Args:
        dm: Base model (e.g., a linear model mapping features to logits).
        opt_inner: Inner-loop optimizer (e.g., SGD or Adam).
        tasks: Sequence of (features, targets) pairs for adaptation.
        num_gd_steps: Number of gradient descent steps per task.

    Returns:
        Scalar tensor containing the total inner-loop loss across tasks.
    """
    if higher is None:
        raise RuntimeError(
            "The 'higher' package is required for meta-learning. "
            "Install it via 'pip install higher'."
        )

    total_loss = torch.tensor(0.0, dtype=torch.float32)

    for features_t, y_t in tasks:
        features_t = features_t.detach()
        y_t = y_t.detach()

        with higher.innerloop_ctx(
            dm,
            opt_inner,
            copy_initial_weights=False,
            track_higher_grads=True,
        ) as (mod, opt):
            for _ in range(num_gd_steps):
                logits = mod(features_t)
                loss = classification_logistic_loss(logits, y_t)

                opt.zero_grad()
                loss.backward(retain_graph=True)
                opt.step(loss)

            # Accumulate loss for logging / teaching purposes
            total_loss = total_loss + loss.detach()

    return total_loss

@dataclass
class VAEConfig:
    """Configuration for a Bernoulli VAE experiment.

    Attributes:
        latent_dim: Dimensionality of the latent space.
        learning_rate: Learning rate for the optimizer.
        batch_size: Training batch size.
        num_epochs: Number of training epochs.
        seed: Random seed for reproducibility.
    """

    latent_dim: int
    learning_rate: float
    batch_size: int
    num_epochs: int
    seed: int = 42


def sample_gaussian(mean: Tensor, logvar: Tensor) -> Tensor:
    """Sample from a diagonal Gaussian using the reparameterization trick.

    Given q(z | x) = N(z; μ, diag(σ²)), where logvar = log σ², we draw:

        ε ~ N(0, I),
        z = μ + σ ⊙ ε.

    Args:
        mean: Tensor of shape (batch_size, latent_dim) with μ.
        logvar: Tensor of shape (batch_size, latent_dim) with log σ².

    Returns:
        Tensor of shape (batch_size, latent_dim) containing samples z.
    """
    std = torch.exp(0.5 * logvar)
    eps = torch.randn_like(std)
    return mean + std * eps


class VAE(nn.Module):
    """Bernoulli VAE with explicit negative ELBO computation.

    This class is deliberately minimal and designed to be used as a teaching
    reference. In the CS182 homework, you typically already have a VAE class
    defined; you can copy the `negative_elbo_bound` method into that class
    rather than using this skeleton directly.
    """

    def __init__(self, encoder: nn.Module, decoder: nn.Module):
        """Initialize the VAE with an encoder and decoder.

        Args:
            encoder: Module mapping x to (mean, logvar) of q(z | x).
            decoder: Module mapping z to logits of p(x | z).
        """
        super().__init__()
        self.enc = encoder
        self.dec = decoder

    def encode(self, x: Tensor) -> Tuple[Tensor, Tensor]:
        """Encode x into mean and log-variance of q(z | x)."""
        return self.enc(x)

    def decode(self, z: Tensor) -> Tensor:
        """Decode latent z into logits for Bernoulli p(x | z)."""
        return self.dec(z)

    def negative_elbo_bound(self, x: Tensor) -> Tuple[Tensor, Dict[str, Tensor]]:
        """Compute the negative ELBO for a Bernoulli VAE.

        ELBO(x) = E_q(z|x)[log p(x|z)] - KL(q(z|x) || p(z)),
        where:
            - p(z) = N(0, I)
            - p(x|z) is Bernoulli with logits given by decoder(z).

        We return the negative ELBO:

            NELBO = KL(q(z|x) || p(z)) - E_q[log p(x|z)]
                  = KL + reconstruction_loss

        Args:
            x: Input batch of images, normalized in [0, 1],
               shape (B, C, H, W) or (B, D).

        Returns:
            nelbo: Scalar tensor to minimize.
            stats: Dict with detached components {"nelbo", "kl", "rec"}.
        """
        if ut is None:
            raise RuntimeError(
                "Utility module 'codebase.utils' could not be imported. "
                "Adapt the import at the top of this file to match your "
                "project structure."
            )

        # Encode approximate posterior q(z | x)
        mean, logvar = self.encode(x)

        # Sample z via reparameterization
        z = sample_gaussian(mean, logvar)

        # KL(q || p): E_q[log q(z|x) - log p(z)]
        log_qz = ut.log_normal(z, mean, logvar)  # type: ignore[attr-defined]
        log_pz = ut.log_normal_standard(z)       # type: ignore[attr-defined]
        kl = torch.mean(log_qz - log_pz)

        # Reconstruction term: -E_q[log p(x|z)]
        logits_x = self.decode(z)
        log_pxz = ut.log_bernoulli_with_logits(x, logits_x)  # type: ignore[attr-defined]
        rec = -torch.mean(log_pxz)

        nelbo = kl + rec

        stats = {
            "nelbo": nelbo.detach(),
            "kl": kl.detach(),
            "rec": rec.detach(),
        }
        return nelbo, stats
