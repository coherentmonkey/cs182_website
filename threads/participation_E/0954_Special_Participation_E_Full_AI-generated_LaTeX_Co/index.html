I used Jameson's lecture note converter to convert the handwritten course notes to LaTeX for all 27 lectures. I can't find his post otherwise I'd mention it to give him credit. Basically it uses the following Python code and combines the lecture slides with the YouTube audio transcript. I slightly modified his code to not paste the API key directly in the script and instead use dotenv: I also elegantly formatted them in a more note-like structure with a table of contents, custom headers and section numbers corresponding to the lecture number. from io import BytesIO
from google import genai
from google.genai.types import Part
from pdf2image import convert_from_path
from tqdm import tqdm
from dotenv import load_dotenv
import os

load_dotenv()

def pdf_to_images(pdf_path):
    print(f"Converting PDF to images: {pdf_path}")
    images = convert_from_path(pdf_path, dpi=300)
    print(f"Successfully converted {len(images)} pages to images")
    
    return images

def read_transcript(txt_path):
    with open(txt_path, 'r', encoding='utf-8') as f:
        content = f.read()
    print(f"Successfully read transcript from {txt_path}")
    
    return content

def image_to_bytes(image):
    buffered = BytesIO()
    image.save(buffered, format="PNG")

    return buffered.getvalue()

def call_gemini_api(images, transcript, GEMINI_API_KEY):
    client = genai.Client(api_key=GEMINI_API_KEY)
    
    prompt = """
    Attached are the lecture video transcript and slides for CS 182, Deep Neural Networks. 
    I want you to:

    1. clean up the transcript, adjusting any typos
    2. think about how the transcript aligns with the notes
    3. create comprehensive notes that cover the entire lecture, such that I can read it and obtain the same information as watching the lecture
    
    Output the notes ONLY in academic latex format. 
    Do not include any other text or formatting. 
    As much content from the slides as possible should be included in the notes, supplemented by the transcript. 
    Do not include images in the output, and don't include a date.

    Here is the transcript:

    """ + transcript
    
    print("Attaching images to prompt...")
    contents = [prompt]
    for image in tqdm(images):
        image_bytes = image_to_bytes(image)
        contents.append(
            Part.from_bytes(
                data=image_bytes,
                mime_type="image/png"
            )
        )
    
    print("Calling Gemini API...")

    response = client.models.generate_content(
        model="gemini-2.5-pro",
        contents=contents
    )
    result = response.text
    print("Successfully received response from Gemini API")

    return result

def clean_latex_output(content):
    content = content.lstrip()
    if content.startswith("```latex"):
        content = content[8:].lstrip()
    elif content.startswith("```"):
        content = content[3:].lstrip()
    
    content = content.rstrip()
    if content.endswith("```"):
        content = content[:-3].rstrip()
    
    return content

def save_latex_output(content, output_path):
    content = clean_latex_output(content)
    
    with open(output_path, 'w', encoding='utf-8') as f:
        f.write(content)
    print(f"Successfully saved LaTeX output to {output_path}")


if __name__ == "__main__":
    lec = "4"

    # List of PDF files for this lecture (can be one or multiple)
    PDF_FILES = [
        f"pdfs/{lec}.pdf"
    ]

    TXT_FILE = "transcripts/" + lec + ".txt"
    OUTPUT_FILE = "latex/" + lec + ".tex"
    GEMINI_API_KEY = os.environ["GOOGLE_API_KEY"]

    # Convert all PDFs to images and combine them
    all_images = []
    for pdf_file in PDF_FILES:
        images = pdf_to_images(pdf_file)
        all_images.extend(images)

    print(f"Total images from all PDFs: {len(all_images)}")

    transcript = read_transcript(TXT_FILE)

    latex_content = call_gemini_api(all_images, transcript, GEMINI_API_KEY)

    save_latex_output(latex_content, OUTPUT_FILE)
