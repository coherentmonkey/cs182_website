import numpy as np
from deeplearning.layers import affine_forward, affine_backward, softmax_loss
from deeplearning.layer_utils import affine_relu_forward, affine_relu_backward

class TwoLayerNet(object):
    """
    A two-layer fully-connected neural network with ReLU nonlinearity and
    softmax loss. Architecture: affine - relu - affine - softmax
    """

    def __init__(self, input_dim=3 * 32 * 32, hidden_dim=100,
                 num_classes=10, weight_scale=1e-3, reg=0.0):
        self.params = {}
        self.reg = reg

        ############################################################################
        # Initialize weights and biases                                            #
        ############################################################################
        # First layer: D -> H
        self.params['W1'] = weight_scale * np.random.randn(input_dim, hidden_dim)
        self.params['b1'] = np.zeros(hidden_dim)

        # Second layer: H -> C
        self.params['W2'] = weight_scale * np.random.randn(hidden_dim, num_classes)
        self.params['b2'] = np.zeros(num_classes)
        ############################################################################
        #                            END OF CODE                                   #
        ############################################################################

    def loss(self, X, y=None):
        """
        Compute loss and gradients for a minibatch of data.
        """
        W1, b1 = self.params['W1'], self.params['b1']
        W2, b2 = self.params['W2'], self.params['b2']

        # Forward pass
        ############################################################################
        # Compute class scores                                                      #
        ############################################################################
        # First layer: affine -> relu
        out_hidden, cache_hidden = affine_relu_forward(X, W1, b1)

        # Second layer: affine
        scores, cache_scores = affine_forward(out_hidden, W2, b2)
        ############################################################################
        #                           END OF CODE                                    #
        ############################################################################

        # If test mode, just return the scores
        if y is None:
            return scores

        ############################################################################
        # Compute softmax loss + backward                                          #
        ############################################################################

        # Loss (data loss + regularization loss)
        loss, dscores = softmax_loss(scores, y)

        # Add regularization: 0.5 * reg * ∑ W^2
        loss += 0.5 * self.reg * (np.sum(W1 * W1) + np.sum(W2 * W2))

        # Backward pass
        grads = {}

        # Backprop through second layer
        dhidden, dW2, db2 = affine_backward(dscores, cache_scores)

        # Backprop through first layer (affine + relu)
        dX, dW1, db1 = affine_relu_backward(dhidden, cache_hidden)

        # Add reg gradients (only weights)
        dW2 += self.reg * W2
        dW1 += self.reg * W1

        # Store grads
        grads['W1'] = dW1
        grads['b1'] = db1
        grads['W2'] = dW2
        grads['b2'] = db2
        ############################################################################
        #                           END OF CODE                                    #
        ############################################################################

        return loss, grads

class FullyConnectedNet(object):
    """
    Fully-connected network: {affine - relu} x (L-1) - affine - softmax
    """

    def __init__(self, hidden_dims, input_dim=3 * 32 * 32,
                 num_classes=10, dropout=0, use_batchnorm=False,
                 reg=0.0, weight_scale=1e-2, dtype=np.float32, seed=None):

        self.use_batchnorm = use_batchnorm
        self.use_dropout = dropout > 0
        self.reg = reg
        self.num_layers = 1 + len(hidden_dims)
        self.dtype = dtype
        self.params = {}

        ############################################################################
        # Parameter initialization                                                 #
        ############################################################################

        layer_input_dim = input_dim

        # Hidden layers
        for i, hidden_dim in enumerate(hidden_dims):
            self.params['W%d' % (i+1)] = weight_scale * np.random.randn(layer_input_dim, hidden_dim)
            self.params['b%d' % (i+1)] = np.zeros(hidden_dim)
            layer_input_dim = hidden_dim

        # Final output layer
        self.params['W%d' % self.num_layers] = weight_scale * np.random.randn(layer_input_dim, num_classes)
        self.params['b%d' % self.num_layers] = np.zeros(num_classes)

        ############################################################################

        # Cast parameters
        for k, v in self.params.items():
            self.params[k] = v.astype(dtype)

        # Ignore dropout and batchnorm for now

    def loss(self, X, y=None):
        """
        Compute loss and gradient.
        """
        X = X.astype(self.dtype)
        mode = 'test' if y is None else 'train'

        caches = {}
        out = X

        ############################################################################
        # Forward pass                                                             #
        ############################################################################

        # Forward through hidden layers
        for i in range(1, self.num_layers):
            Wi = self.params['W%d' % i]
            bi = self.params['b%d' % i]
            out, caches[i] = affine_relu_forward(out, Wi, bi)

        # Final affine layer
        W_last = self.params['W%d' % self.num_layers]
        b_last = self.params['b%d' % self.num_layers]
        scores, caches[self.num_layers] = affine_forward(out, W_last, b_last)

        # Test mode: return scores only
        if y is None:
            return scores

        ############################################################################
        # Backward pass                                                            #
        ############################################################################

        grads = {}
        loss, dscores = softmax_loss(scores, y)

        # Add regularization to loss
        for i in range(1, self.num_layers+1):
            Wi = self.params['W%d' % i]
            loss += 0.5 * self.reg * np.sum(Wi * Wi)

        # Backprop through final layer
        dout, dW, db = affine_backward(dscores, caches[self.num_layers])
        grads['W%d' % self.num_layers] = dW + self.reg * W_last
        grads['b%d' % self.num_layers] = db

        # Backprop through hidden layers (reverse order)
        for i in reversed(range(1, self.num_layers)):
            Wi = self.params['W%d' % i]
            dout, dW, db = affine_relu_backward(dout, caches[i])
            grads['W%d' % i] = dW + self.reg * Wi
            grads['b%d' % i] = db

        return loss, grads

import numpy as np


def affine_forward(x, w, b):
    """
    Computes the forward pass for an affine (fully-connected) layer.
    """
    # (N, d1, d2, ..., dk)
    N = x.shape[0]

    # Reshape x into (N, D)
    x_row = x.reshape(N, -1)

    # Compute output: (N, M)
    out = x_row.dot(w) + b  # b broadcasted

    # Values needed for backward pass
    cache = (x, w, b)

    return out, cache



def affine_backward(dout, cache):
    """
    Computes the backward pass for an affine layer.
    """
    x, w, b = cache
    N = x.shape[0]

    # Reshape x to (N, D)
    x_row = x.reshape(N, -1)

    # Gradients
    dx = dout.dot(w.T)                    # (N, D)
    dx = dx.reshape(x.shape)              # back to (N, d1, ..., dk)

    dw = x_row.T.dot(dout)                # (D, M)

    db = np.sum(dout, axis=0)             # (M,)

    return dx, dw, db



def relu_forward(x):
    """
    Computes the forward pass for a layer of rectified linear units (ReLUs).
    """
    # Apply ReLU elementwise
    out = np.maximum(0, x)

    # Store x for backward pass
    cache = x
    return out, cache



def relu_backward(dout, cache):
    """
    Computes the backward pass for a layer of rectified linear units (ReLUs).
    """
    x = cache
    
    # Gradient flows where x > 0
    dx = dout * (x > 0)
    
    return dx


def svm_loss(x, y):
    """
    Computes the loss and gradient using for multiclass SVM classification.

    Inputs:
    - x: Input data, of shape (N, C) where x[i, j] is the score for the jth class
      for the ith input.
    - y: Vector of labels, of shape (N,) where y[i] is the label for x[i] and
      0 <= y[i] < C

    Returns a tuple of:
    - loss: Scalar giving the loss
    - dx: Gradient of the loss with respect to x
    """
    N = x.shape[0]
    correct_class_scores = x[np.arange(N), y]
    margins = np.maximum(0, x - correct_class_scores[:, np.newaxis] + 1.0)
    margins[np.arange(N), y] = 0
    loss = np.sum(margins) / N
    num_pos = np.sum(margins > 0, axis=1)
    dx = np.zeros_like(x)
    dx[margins > 0] = 1
    dx[np.arange(N), y] -= num_pos
    dx /= N
    return loss, dx


def softmax_loss(x, y):
    """
    Computes the loss and gradient for softmax classification.

    Inputs:
    - x: Input data, of shape (N, C) where x[i, j] is the score for the jth class
      for the ith input.
    - y: Vector of labels, of shape (N,) where y[i] is the label for x[i] and
      0 <= y[i] < C

    Returns a tuple of:
    - loss: Scalar giving the loss
    - dx: Gradient of the loss with respect to x
    """
    probs = np.exp(x - np.max(x, axis=1, keepdims=True))
    probs /= np.sum(probs, axis=1, keepdims=True)
    N = x.shape[0]
    loss = -np.sum(np.log(probs[np.arange(N), y])) / N
    dx = probs.copy()
    dx[np.arange(N), y] -= 1
    dx /= N
    return loss, dx
